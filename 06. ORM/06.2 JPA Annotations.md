## 🏷️ Entity Mapping Cơ Bản

```java
import jakarta.persistence.*;  // Jakarta EE (Spring Boot 3+)
// import javax.persistence.*;  // cũ hơn (Spring Boot 2.x)

@Entity                          // ① Đánh dấu class là JPA Entity
@Table(                          // ② Map vào table cụ thể (tùy chọn)
    name = "users",
    indexes = {
        @Index(name = "idx_email", columnList = "email"),
        @Index(name = "idx_status", columnList = "status")
    },
    uniqueConstraints = {
        @UniqueConstraint(columnNames = {"email"})
    }
)
@Getter @Setter  // Lombok
@NoArgsConstructor
@AllArgsConstructor
public class User {

    @Id                                              // ③ Primary Key
    @GeneratedValue(strategy = GenerationType.IDENTITY)  // ④ Auto-increment
    private Long id;

    @Column(                                         // ⑤ Column mapping
        name = "full_name",        // tên cột trong DB (default: field name)
        nullable = false,          // NOT NULL constraint
        length = 100               // VARCHAR(100)
    )
    private String name;

    @Column(unique = true, nullable = false, length = 255)
    private String email;

    @Column(name = "age")
    private Integer age;

    @Enumerated(EnumType.STRING)                     // ⑥ Enum → VARCHAR
    @Column(nullable = false)
    private UserStatus status = UserStatus.ACTIVE;

    @Lob                                             // ⑦ Large Object (TEXT/CLOB)
    @Column(columnDefinition = "TEXT")
    private String description;

    @Transient                                       // ⑧ Không map vào DB
    private String temporaryToken;

    @CreationTimestamp                               // ⑨ Hibernate: tự set khi create
    @Column(updatable = false)
    private LocalDateTime createdAt;

    @UpdateTimestamp                                 // Tự update mỗi lần save
    private LocalDateTime updatedAt;

    @Version                                         // ⑩ Optimistic locking
    private Long version;
}
```

## 🔑 GenerationType Strategies

```java
// IDENTITY — Database auto-increment (MySQL, PostgreSQL SERIAL)
@GeneratedValue(strategy = GenerationType.IDENTITY)
private Long id;

// SEQUENCE — Database sequence (PostgreSQL, Oracle) — tốt hơn IDENTITY cho batch
@GeneratedValue(strategy = GenerationType.SEQUENCE, generator = "user_seq")
@SequenceGenerator(name = "user_seq", sequenceName = "users_id_seq", allocationSize = 50)
private Long id;

// UUID — dùng cho distributed systems
@Id
@GeneratedValue(strategy = GenerationType.UUID)
private UUID id;

// Hoặc tự gen
@Id
private String id = UUID.randomUUID().toString();
```

## 🏛️ Embedded Objects

```java
// Tái sử dụng group of columns trong nhiều entities
@Embeddable
public class Address {
    @Column(name = "street")
    private String street;

    @Column(name = "city")
    private String city;

    @Column(name = "country", length = 2)
    private String countryCode;
}

@Entity
public class User {
    @Id @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    @Embedded                              // Nhúng Address vào bảng users
    @AttributeOverrides({                  // Override column names nếu cần
        @AttributeOverride(name = "street", column = @Column(name = "home_street")),
        @AttributeOverride(name = "city",   column = @Column(name = "home_city"))
    })
    private Address homeAddress;

    @Embedded
    @AttributeOverrides({
        @AttributeOverride(name = "street", column = @Column(name = "work_street")),
        @AttributeOverride(name = "city",   column = @Column(name = "work_city"))
    })
    private Address workAddress;
}
```

## 🔄 Entity Lifecycle & Callbacks

```java
@Entity
public class AuditableEntity {

    @PrePersist    // Trước khi INSERT
    protected void onCreate() {
        this.createdAt = LocalDateTime.now();
        this.createdBy = SecurityUtils.getCurrentUser();
    }

    @PreUpdate     // Trước khi UPDATE
    protected void onUpdate() {
        this.updatedAt = LocalDateTime.now();
    }

    @PostLoad      // Sau khi load từ DB
    protected void afterLoad() {
        // Decrypt sensitive fields, etc.
    }
}

// Tốt hơn: Dùng @EntityListeners (Spring Data)
@Entity
@EntityListeners(AuditingEntityListener.class)
public class User {
    @CreatedDate
    @Column(updatable = false)
    private LocalDateTime createdAt;

    @LastModifiedDate
    private LocalDateTime updatedAt;

    @CreatedBy
    @Column(updatable = false)
    private String createdBy;

    @LastModifiedBy
    private String lastModifiedBy;
}
```

## 💡 Best Practices

```java
// ✅ Luôn implement equals() và hashCode() dựa trên natural ID
@Entity
public class Product {
    @Id @GeneratedValue
    private Long id;

    @Column(unique = true, nullable = false)
    private String sku;  // natural ID

    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (!(o instanceof Product p)) return false;
        return sku != null && sku.equals(p.sku);
    }

    @Override
    public int hashCode() {
        return Objects.hash(sku);
    }
}

// ✅ Dùng BaseEntity để tái sử dụng common fields
@MappedSuperclass  // không tạo table riêng, embed fields vào subclasses
@EntityListeners(AuditingEntityListener.class)
public abstract class BaseEntity {
    @Id @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    @CreatedDate @Column(updatable = false)
    private LocalDateTime createdAt;

    @LastModifiedDate
    private LocalDateTime updatedAt;
}

@Entity
public class User extends BaseEntity {
    // Chỉ cần thêm fields riêng
    private String name;
    private String email;
}
```
